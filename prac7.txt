father(X,Y) :-
    parent(X,Y),

mother(X,Y) :-
    parent(X,Y),
    female(X).

child(X,Y) :-
    parent(Y,X).

sibling(X,Y) :-
    parent(Z,X),
    parent(Z,Y),
    X \= Y.

grandparent(X,Y) :-
    parent(X,Z),
    parent(Z,Y).

ancestor(X,Y) :-
    parent(X,Y).

    parent(X,Z),
